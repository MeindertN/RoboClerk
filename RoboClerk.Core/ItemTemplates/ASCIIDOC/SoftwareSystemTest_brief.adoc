using RoboClerk;
using System.Linq;
using System;

TraceEntity te = SourceTraceEntity;

bool showResults = GetTagParameter("showResults","no").ToUpper() == "TRUE";

string softwareRequirementName = TraceabilityAnalysis.GetTitleForTraceEntity("SoftwareRequirement");

string CreateTopRow()
{
	if (showResults)
		return $"| *{te.Name} ID* | *{te.Name} Description* | *Linked {softwareRequirementName}s*  | *Actual Result* | *Timestamp* \n";
	else
		return $"| *{te.Name} ID* | *{te.Name} Description* | *Linked {softwareRequirementName}s* \n";
}

string CreateRows()
{
	System.Text.StringBuilder sb = new System.Text.StringBuilder();
	foreach( var item in Items)
	{
		SoftwareSystemTestItem st = (SoftwareSystemTestItem)item;
		AddTrace(st.ItemID);

		//find all requirements traced to this test and get them ready for display
		System.Text.StringBuilder reqs = new System.Text.StringBuilder();
		foreach( var req in GetLinkedItems(st, ItemLinkType.Tests) )
		{
			if (reqs.Length > 0)
				reqs.Append(", ");
			reqs.Append(GetItemLinkString(req));
		}

		if (showResults)
		{
			string actualResult = "*NO RESULT*";
			string timestamp = "N/A";
			if (st.TestCaseToUnitTest)
			{
				var unitTests = GetLinkedItems(st, ItemLinkType.UnitTest);
				bool passed = unitTests.Any();
				bool resultsFound = false;
				DateTime maxTime = DateTime.MinValue;

				foreach( var test in unitTests )
				{
					var resultItems = GetLinkedItems(test, ItemLinkType.Result).OfType<TestResult>();
					var testResult = resultItems.OrderByDescending(r => r.ExecutionTime).FirstOrDefault();
					if (testResult != null)
					{
						resultsFound = true;
						if (testResult.ExecutionTime > maxTime)
						{
							maxTime = testResult.ExecutionTime;
						}
					}
					passed = passed && (testResult != null && testResult.ResultStatus == TestResultStatus.PASS);
				}
				if (resultsFound && unitTests.Any())
				{
					actualResult = passed ? "PASS" : "FAIL";
					timestamp = maxTime != DateTime.MinValue ? maxTime.ToString(@"yyyy-MM-ddTHH:mm:ss") : "N/A";
				}
			}
			else
			{
				var resultItems = GetLinkedItems(st, ItemLinkType.Result).OfType<TestResult>();
				var testResult = resultItems.OrderByDescending(r => r.ExecutionTime).FirstOrDefault();
				actualResult = testResult != null ? testResult.ResultStatus.ToString() : "*NO RESULT*";
				timestamp = testResult != null ? testResult.ExecutionTime.ToString(@"yyyy-MM-ddTHH:mm:ss") : "N/A";
			}
			sb.Append($"| {GetItemLinkString(st)} | {st.ItemTitle} | {(reqs.Length==0?"Missing":reqs.ToString())} | {actualResult} | {timestamp} \n\n");
		}
		else
			sb.Append($"| {GetItemLinkString(st)} | {st.ItemTitle} | {(reqs.Length==0?"Missing":reqs.ToString())} \n\n");
	}
	return sb.ToString();
